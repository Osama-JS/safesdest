الموقع:

الخصائص:
المعامل النوع الوصف مطلوب
$to string|array عنوان البريد الإلكتروني للمستلم مطلوب
$subject string موضوع البريد الإلكتروني مطلوب
$template string اسم القالب المستخدم مطلوب
$data array البيانات المرسلة للقالب اختياري

Services - خدمات الإيميل

EmailNotificationService
خدمة

الخدمة الأساسية لإدارة وإرسال الإشعارات عبر البريد الإلكتروني.

الموقع:
app/Services/EmailNotificationService.php
الدوال الأساسية:
sendNotification()

إرسال إشعار فوري عبر البريد الإلكتروني

public function sendNotification( $to, $subject, $template, $data = [], $attachments = [], $priority = 'normal' ) { // التحقق من صحة البيانات $this->validateEmailData($to, $subject, $template); // تسجيل محاولة الإرسال $logId = $this->logNotificationAttempt($to, $subject, $template); try { // إرسال البريد الإلكتروني Mail::to($to)->send(new NotificationMail( $subject, $template, $data, $attachments )); // تسجيل نجاح الإرسال $this->logNotificationSuccess($logId); return true; } catch (Exception $e) { // تسجيل فشل الإرسال $this->logNotificationFailure($logId, $e->getMessage()); throw $e; } }
queueNotification()

إضافة إشعار إلى قائمة الانتظار للإرسال اللاحق

public function queueNotification( $to, $subject, $template, $data = [], $attachments = [], $priority = 'normal', $delay = null ) { $job = new SendEmailNotificationJob( $to, $subject, $template, $data, $attachments, $priority ); // تحديد الطابور حسب الأولوية $queue = $this->getQueueByPriority($priority); if ($delay) { return dispatch($job)->onQueue($queue)->delay($delay); } return dispatch($job)->onQueue($queue); }
sendBulkNotifications()

إرسال إشعارات جماعية لعدة مستلمين

public function sendBulkNotifications( array $recipients, $subject, $template, $data = [], $attachments = [], $priority = 'normal' ) { $jobs = []; foreach ($recipients as $recipient) { $personalizedData = $this->personalizeData($data, $recipient); $jobs[] = new SendEmailNotificationJob( $recipient['email'], $subject, $template, $personalizedData, $attachments, $priority ); } // إرسال جميع الوظائف دفعة واحدة Bus::batch($jobs) ->then(function (Batch $batch) { // تسجيل نجاح الإرسال الجماعي Log::info('Bulk email notifications sent successfully', [ 'batch_id' => $batch->id, 'total_jobs' => $batch->totalJobs ]); }) ->catch(function (Batch $batch, Throwable $e) { // تسجيل فشل الإرسال الجماعي Log::error('Bulk email notifications failed', [ 'batch_id' => $batch->id, 'error' => $e->getMessage() ]); }) ->dispatch(); return $jobs; }

Mail Classes - فئات البريد

NotificationMail
قالب

الفئة الأساسية لإنشاء وإرسال رسائل البريد الإلكتروني.

الموقع:
app/Mail/NotificationMail.php
البنية الأساسية:
class NotificationMail extends Mailable { use Queueable, SerializesModels; public $subject; public $template; public $data; public $attachmentFiles; public function __construct($subject, $template, $data = [], $attachments = []) { $this->subject = $subject; $this->template = $template; $this->data = $data; $this->attachmentFiles = $attachments; } public function build() { $mail = $this->subject($this->subject) ->view("emails.{$this->template}", $this->data); // إضافة المرفقات if (!empty($this->attachmentFiles)) { foreach ($this->attachmentFiles as $attachment) { if (is_array($attachment)) { $mail->attach($attachment['path'], [ 'as' => $attachment['name'] ?? null, 'mime' => $attachment['mime'] ?? null, ]); } else { $mail->attach($attachment); } } } return $mail; } }
TransactionReceipt
قالب

فئة متخصصة لإرسال إيصالات المعاملات المالية.

class TransactionReceipt extends Mailable { public $transaction; public $user; public function __construct($transaction, $user) { $this->transaction = $transaction; $this->user = $user; } public function build() { return $this->subject('إيصال معاملة - ' . $this->transaction->reference_number) ->view('emails.transaction_receipt') ->with([ 'transaction' => $this->transaction, 'user' => $this->user, 'company_name' => config('app.name') ]); } }
