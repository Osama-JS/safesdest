// تحسينات مقترحة لدالة getData() public function getData(Request $request) { $query = Customs_Clearance::with(['customer', 'user', 'formTemplate', 'clearanceAgent']) ->select(['id', 'status', 'total_price', 'customer_id', 'user_id', 'form_template_id', 'clearance_agent_id', 'created_at']); // إضافة الفلاتر if ($request->filled('status')) { $query->where('status', $request->status); } if ($request->filled('owner_type')) { if ($request->owner_type === 'customer') { $query->whereNotNull('customer_id'); } else { $query->whereNotNull('user_id'); } } return DataTables::of($query) ->addColumn('owner', function($row) { return $row->customer ? $row->customer->name : $row->user->name; }) ->addColumn('owner_type', function($row) { return $row->customer ? 'Customer' : 'Admin'; }) ->addColumn('actions', function($row) { return view('admin.customs-clearances.actions', compact('row')); }) ->rawColumns(['actions', 'status']) ->make(true); }
