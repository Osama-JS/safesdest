⚙التنفيذ التقني المفصل

1. تحسين دالة طلب الصلاحيات الرئيسية:

safedest_driver/lib/services/location_service.dart
// Request location permission with enhanced handling Future _requestLocationPermission() async { try { debugPrint('🔍 Starting comprehensive location permission check...'); // Step 1: Check if location services are enabled bool serviceEnabled = await Geolocator.isLocationServiceEnabled(); if (!serviceEnabled) { // Try to prompt user to enable location services await Geolocator.openLocationSettings(); await Future.delayed(const Duration(seconds: 1)); serviceEnabled = await Geolocator.isLocationServiceEnabled(); } // Step 2: Check permission using permission_handler PermissionStatus permissionStatus = await Permission.location.status; // Step 3: Request permission if needed if (permissionStatus.isDenied) { permissionStatus = await Permission.location.request(); } // Step 4: Handle permanently denied if (permissionStatus.isPermanentlyDenied) { await openAppSettings(); await Future.delayed(const Duration(seconds: 2)); permissionStatus = await Permission.location.status; } // Step 5: Double-check with Geolocator if (permissionStatus.isGranted) { LocationPermission geoPermission = await Geolocator.checkPermission(); if (geoPermission == LocationPermission.denied) { geoPermission = await Geolocator.requestPermission(); } return geoPermission == LocationPermission.whileInUse || geoPermission == LocationPermission.always; } return false; } catch (e) { debugPrint('💥 Location permission error: $e'); return false; } }

2. تحسين دالة طلب الصلاحية للتطبيق:

// Request GPS permission only (for app startup) with enhanced retry logic Future requestGPSPermissionOnly() async { try { // First attempt with comprehensive permission check final hasPermission = await _requestLocationPermission(); if (hasPermission) { return true; } // If first attempt failed, try alternative approach debugPrint('🔄 First attempt failed, trying alternative permission request...'); // Try using permission_handler directly final status = await Permission.location.request(); if (status.isGranted) { debugPrint('✅ GPS permission granted via alternative method'); return true; } else if (status.isPermanentlyDenied) { debugPrint('❌ GPS permission permanently denied'); return false; } return false; } catch (e) { debugPrint('💥 GPS permission request error: $e'); return false; } }

3. إضافة دالة التشخيص المفصل:

// Check detailed permission status for debugging Future> getDetailedPermissionStatus() async { try { final serviceEnabled = await Geolocator.isLocationServiceEnabled(); final permissionHandlerStatus = await Permission.location.status; final geolocatorPermission = await Geolocator.checkPermission(); final status = { 'serviceEnabled': serviceEnabled, 'permissionHandlerStatus': permissionHandlerStatus.toString(), 'geolocatorPermission': geolocatorPermission.toString(), 'isGranted': permissionHandlerStatus.isGranted, 'isDenied': permissionHandlerStatus.isDenied, 'isPermanentlyDenied': permissionHandlerStatus.isPermanentlyDenied, }; debugPrint('📊 Detailed permission status: $status'); return status; } catch (e) { debugPrint('💥 Error getting permission status: $e'); return {'error': e.toString()}; } }

4. تحسين SplashScreen مع محاولات متعددة:

// Request GPS permission during app startup with enhanced handling Future _requestGPSPermission(LocationService locationService) async { try { // Try multiple times with different approaches bool hasPermission = false; int attempts = 0; const maxAttempts = 3; while (!hasPermission && attempts < maxAttempts) { attempts++; debugPrint('🔄 SplashScreen: Permission attempt $attempts/$maxAttempts'); // Request permission with enhanced method hasPermission = await locationService.requestGPSPermissionOnly(); if (!hasPermission && attempts < maxAttempts) { await Future.delayed(const Duration(seconds: 1)); } } if (hasPermission) { // Try to get initial location to verify everything works final response = await locationService.sendLocationManually(); if (response.isSuccess) { debugPrint('✅ SplashScreen: Location access verified successfully'); } } } catch (e) { debugPrint('💥 SplashScreen: GPS permission request error: $e'); } }
