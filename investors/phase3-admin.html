منطق إنشاء حساب المستثمر

يتم إنشاء مستثمر جديد عبر خطوتين: إنشاء حساب المستخدم ثم تفعيل وضع المستثمر وإنشاء محفظة الاستثمار والعقد.

// InvestorController@store public function store(StoreInvestorRequest $request) { DB::transaction(function() use ($request) { // 1. إنشاء حساب المستخدم $user = User::create([ 'name' => $request->name, 'email' => $request->email, 'phone' => $request->phone, 'password' => bcrypt($request->password), 'investor' => true, 'status' => 'active', ]); // 2. إنشاء محفظة الاستثمار InvestorWallet::create([ 'user_id' => $user->id, 'status' => true, ]); // 3. إنشاء محفظة العمولات الشخصية (user_wallets) UserWallet::create([ 'user_id' => $user->id, 'user_type' => 'user', 'status' => true, ]); // 4. إنشاء عقد الاستثمار InvestmentContract::create([ 'user_id' => $user->id, 'contract_type' => $request->contract_type, 'commission_type' => $request->commission_type, 'commission_value' => $request->commission_value, 'start_date' => $request->start_date, 'end_date' => $request->end_date, 'filter_customer_ids' => $request->filter_customer_ids, 'min_commission_threshold' => $request->min_commission_threshold, 'status' => 'active', 'created_by' => auth()->id(), ]); }); return redirect()->route('admin.investors.index') ->with('success', 'تم إنشاء حساب المستثمر بنجاح'); }

💰 تحميل محفظة الاستثمار (Admin)

آلية إيداع الأموال في محفظة الاستثمار

يقوم Admin بإيداع مبالغ الاستثمار في محفظة المستثمر. يُسجَّل كل إيداع في investor_wallet_transactions كعملية credit.

// InvestorWalletController@deposit public function deposit(Request $request, User $investor) { $request->validate([ 'amount' => 'required|numeric|min:1', 'description' => 'nullable|string|max:500', ]); DB::transaction(function() use ($request, $investor) { $wallet = $investor->investorWallet; $balanceBefore = $wallet->balance; $newBalance = $balanceBefore + $request->amount; InvestorWalletTransaction::create([ 'investor_wallet_id' => $wallet->id, 'transaction_type' => 'credit', 'amount' => $request->amount, 'description' => $request->description ?? 'إيداع من الإدارة', 'performed_by' => auth()->id(), 'balance_after' => $newBalance, ]); }); return back()->with('success', 'تم الإيداع بنجاح'); }
