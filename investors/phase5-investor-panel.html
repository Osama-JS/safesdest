class InvestorDashboardController extends Controller { public function index() { $investor = auth()->user()->load([ 'investorWallet.transactions', 'userWallet.transactions', 'activeInvestmentContract', ]); $investorWallet = $investor->investorWallet; $personalWallet = $investor->userWallet; $contract = $investor->activeInvestmentContract; // إحصائيات سريعة $stats = [ 'investment_balance' => $investorWallet?->balance ?? 0, 'personal_balance' => $personalWallet?->balance ?? 0, 'paid_tasks_count' => Task::where('investor_id', $investor->id)->count(), 'total_commissions' => $personalWallet?->transactions() ->where('transaction_type', 'credit') ->sum('amount') ?? 0, ]; // آخر 5 عمليات مجمعة $recentActivity = InvestorWalletTransaction::where( 'investor_wallet_id', $investorWallet?->id )->latest()->take(5)->get(); return view('investor.dashboard', compact( 'investor', 'stats', 'recentActivity', 'contract' )); } }

✅ قائمة مهام التنفيذ الكاملة

  • إنشاء investor.blade.php كـ Layout رئيسي بقائمة جانبية ديناميكية
  • بناء InvestorDashboardController@index مع كل الإحصائيات
  • بناء صفحة محفظة الاستثمار مع فلترة الحركات
  • بناء صفحة المحفظة الشخصية مع فلترة العمولات
  • بناء صفحة دفع المهام (task_investment) مع فلاتر البحث
  • إنشاء modal تأكيد الدفع مع عرض تفاصيل المهمة والرصيد
  • بناء صفحة المهام المدفوعة مع إمكانية التصدير
  • بناء صفحة الملف الشخصي وتغيير كلمة المرور
  • إضافة AJAX للدفع لتجنب إعادة تحميل الصفحة
  • إضافة رسائل نجاح/خطأ واضحة بعد كل عملية
  • دعم RTL كامل في كل الصفحات
  • اختبار تجربة المستخدم على الأجهزة المحمولة
