// إرسال Token إلى الخادم Future<bool> updateFcmToken(String token) async { final response = await _apiService.post<void>( AppConfig.updateFcmTokenEndpoint, body: {'fcm_token': token}, fromJson: (data) {}, ); return response.success; }

💻 كيفية إرسال الإشعارات من أي Controller في Laravel

🔧 الطريقة الأساسية

<?php use App\Services\NotificationService; use App\Models\Driver; class YourController extends Controller { protected $notificationService; public function __construct() { $this->notificationService = new NotificationService(); } public function sendNotification() { // الحصول على السائق $driver = Driver::find(1); // إرسال إشعار مهمة جديدة $task = (object) [ 'id' => 123, 'pickup' => (object) ['address' => 'الرياض - حي النخيل'], 'delivery' => (object) ['address' => 'الرياض - حي الملز'], 'price' => 50 ]; $result = $this->notificationService->sendNewTaskNotificationToDriver($driver, $task); if ($result['success']) { return response()->json(['message' => 'تم إرسال الإشعار بنجاح']); } else { return response()->json(['error' => $result['message']], 500); } } }

📱 أمثلة عملية لإرسال الإشعارات

// في TaskController public function assignTaskToDriver($taskId, $driverId) { $task = Task::find($taskId); $driver = Driver::find($driverId); // تعيين المهمة للسائق $task->driver_id = $driverId; $task->status = 'assigned'; $task->save(); // إرسال إشعار $notificationService = new NotificationService(); $result = $notificationService->sendNewTaskNotificationToDriver($driver, $task); return response()->json([ 'task_assigned' => true, 'notification_sent' => $result['success'] ]); }

💰 إشعار دفعة مالية

// في PaymentController public function addToWallet($driverId, $amount) { $driver = Driver::find($driverId); // إضافة المبلغ للمحفظة $driver->wallet_balance += $amount; $driver->save(); // إرسال إشعار $notificationService = new NotificationService(); $result = $notificationService->sendPaymentNotificationToDriver( $driver, $amount, 'payment_received' ); return response()->json([ 'wallet_updated' => true, 'new_balance' => $driver->wallet_balance, 'notification_sent' => $result['success'] ]); }

🔄 إشعار تحديث المهمة

// في TaskController public function updateTaskStatus($taskId, $status) { $task = Task::find($taskId); $driver = $task->driver; // تحديث حالة المهمة $task->status = $status; $task->save(); // إرسال إشعار $notificationService = new NotificationService(); $result = $notificationService->sendTaskUpdateNotificationToDriver( $driver, $task, $status ); return response()->json([ 'task_updated' => true, 'notification_sent' => $result['success'] ]); }

📢 إشعار جماعي

// في AdminController public function sendBroadcastNotification(Request $request) { $title = $request->title; $body = $request->body; // الحصول على جميع السائقين النشطين $drivers = Driver::where('is_active', true) ->whereNotNull('fcm_token') ->get(); // إرسال إشعار جماعي $notificationService = new NotificationService(); $result = $notificationService->sendNotificationToMultipleDrivers( $drivers->toArray(), $title, $body, ['type' => 'broadcast', 'timestamp' => now()->toISOString()] ); return response()->json([ 'drivers_count' => $drivers->count(), 'success_count' => $result['success_count'] ?? 0, 'failure_count' => $result['failure_count'] ?? 0 ]); }

🎯 إرسال إشعار مخصص

// إرسال إشعار مخصص بالكامل public function sendCustomNotification($driverId, $title, $body, $data = []) { $driver = Driver::find($driverId); if (!$driver || !$driver->fcm_token) { return response()->json(['error' => 'Driver not found or no FCM token'], 404); } $firebaseService = new FirebaseService(); $result = $firebaseService->sendToDevice( $driver->fcm_token, $title, $body, array_merge($data, [ 'click_action' => 'FLUTTER_NOTIFICATION_CLICK', 'timestamp' => now()->toISOString() ]) ); return response()->json($result); }

استقبال الإشعارات في الخلفية (Background)

// معالج الإشعارات في الخلفية (دالة static) static Future<void> _handleBackgroundMessage(RemoteMessage message) async { debugPrint('Received background message: ${message.messageId}'); // يمكن حفظ الإشعار في قاعدة بيانات محلية } // تسجيل المعالج FirebaseMessaging.onBackgroundMessage(_handleBackgroundMessage);

معالجة النقر على الإشعار

// عند النقر على الإشعار وفتح التطبيق FirebaseMessaging.onMessageOpenedApp.listen(_handleNotificationTap); void _handleNotificationTap(RemoteMessage message) { debugPrint('Notification tapped: ${message.messageId}'); // التنقل حسب نوع الإشعار String? type = message.data['type']; switch (type) { case 'new_task': // الانتقال لصفحة المهام Navigator.pushNamed(context, '/tasks'); break; case 'payment_received': // الانتقال لصفحة المحفظة Navigator.pushNamed(context, '/wallet'); break; default: // الانتقال للصفحة الرئيسية Navigator.pushNamed(context, '/home'); } }
