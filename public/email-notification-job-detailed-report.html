SendEmailNotificationJob - دليل مفصل للاستخدام والتطبيق في Controllers

📋 نظرة عامة على النظام

نظام الإشعارات بالبريد الإلكتروني في التطبيق مبني على Laravel Jobs و Queues لضمان الأداء العالي وعدم التأثير على تجربة المستخدم. يستخدم النظام SendEmailNotificationJob كنواة أساسية لإرسال جميع أنواع الإشعارات.

  • دعم المرفقات المتعددة

🏗️ هيكل SendEmailNotificationJob

الخصائص الأساسية

class SendEmailNotificationJob implements ShouldQueue { use Dispatchable, InteractsWithQueue, Queueable, SerializesModels; public $tries = 3; // عدد المحاولات public $timeout = 120; // مهلة التنفيذ (ثانية) public $backoff = [30, 60, 120]; // فترات الانتظار بين المحاولات protected $emailData; // بيانات البريد الإلكتروني protected $attachments; // المرفقات }

' ريال' ] ]); } }

// في EmailNotificationController public function getStatistics() { $stats = [ 'total_sent' => EmailNotificationLog::sent()->count(), 'total_failed' => EmailNotificationLog::failed()->count(), 'total_pending' => EmailNotificationLog::pending()->count(), 'today_sent' => EmailNotificationLog::sent()->today()->count(), 'this_week' => EmailNotificationLog::sent()->thisWeek()->count(), 'this_month' => EmailNotificationLog::sent()->thisMonth()->count(), 'by_type' => EmailNotificationLog::sent() ->selectRaw('type, COUNT(*) as count') ->groupBy('type') ->pluck('count', 'type'), 'success_rate' => $this->calculateSuccessRate() ]; return response()->json($stats); } private function calculateSuccessRate() { $total = EmailNotificationLog::count(); $sent = EmailNotificationLog::sent()->count(); return $total > 0 ? round(($sent / $total) * 100, 2) : 0; }
